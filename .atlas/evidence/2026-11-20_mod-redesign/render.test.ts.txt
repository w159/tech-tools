type N = { kind?: string; type?: string; props?: Record<string, unknown>; children?: unknown };
const esc = (s: string) => s.replace(/&/g, '&amp;').replace(/</g, '&lt;');
// Element tree -> HTML: Text carries color/backgroundColor/bold; a row Box joins its children, a column Box stacks them.
function html(n: unknown, inherit: Record<string, unknown> = {}): string {
  if (Array.isArray(n)) return n.map((x) => html(x, inherit)).join('');
  if (n === null || n === undefined || typeof n !== 'object') return n == null ? '' : esc(String(n));
  const o = n as N; const p = (o.props ?? {}) as Record<string, unknown>;
  const kind = o.kind ?? o.type;
  if (p.width === 0) return '';
  const bg = (p.backgroundColor as string) ?? (inherit.bg as string);
  if (kind === 'Text') {
    const st = [p.color ? `color:${p.color}` : '', bg ? `background:${bg}` : '', p.bold ? 'font-weight:bold' : '', p.dimColor ? 'opacity:.6' : '', p.underline ? 'text-decoration:underline' : '', p.strikethrough ? 'text-decoration:line-through' : ''].filter(Boolean).join(';');
    const inner = html(p.children, { bg }) + (Array.isArray(o.children) ? html(o.children, { bg }) : '');
    return `<span style="${st}">${inner}</span>`;
  }
  if (kind === 'Input') return `<span style="color:#888;border:1px solid #555;padding:0 6px">${esc(String(p.label ?? ''))} ${esc(String(p.placeholder ?? ''))}</span>`;
  const kids = o.kind ? p.children : o.children;
  const list = Array.isArray(kids) ? kids : kids == null ? [] : [kids];
  const row = p.flexDirection === 'row' || kind === 'Button';
  const style = bg ? `background:${bg}` : '';
  if (row) return `<div style="display:flex;white-space:pre;${style}">${list.map((k) => html(k, { bg })).join('')}</div>`;
  return `<div style="${style}">${list.map((k) => `<div style="white-space:pre">${html(k, { bg })}</div>`).join('')}</div>`;
}
